import json
import os
import sys
import time
from datetime import date, timedelta
from decimal import Decimal
sys.path.insert(0, os.getcwd())
import django
django.setup()
from django.db import connection
from django.test.utils import CaptureQueriesContext
from market.models import Instrument, CandleObservation
from market.state.compute import build_market_state, compute_market_state, ensure_descriptor_definition
from market.state.canonical import canonical_json
from market.tests.test_market_state_context import _seq, observation, event, policy, series

inst = Instrument.objects.get(code="USD_CAD")
cutoff = CandleObservation.objects.order_by("-observed_at").first().observed_at
_seq[0] = 10000
pol = policy("US", "USD")
ser = series(pol, "USD-performance")
for i in range(100):
    at = cutoff - timedelta(hours=10) + timedelta(minutes=i)
    observation(ser, pol, date(2025, 6, 1), Decimal(3)+Decimal(i)/10000, at, at, revision=i)
    observation(ser, pol, date(2025, 7, 1), Decimal(4)+Decimal(i)/10000, at, at, revision=i)
for i in range(200):
    event(pol, "US", cutoff + timedelta(hours=1), cutoff - timedelta(hours=5), key=f"load-{i}")
event(pol, "US", cutoff + timedelta(days=30), cutoff - timedelta(hours=1), key="load-0")
definition = ensure_descriptor_definition()
begin = time.perf_counter()
with CaptureQueriesContext(connection) as queries:
    result = build_market_state(inst, definition, cutoff, ["H1"])
elapsed = time.perf_counter() - begin
snapshot, _ = compute_market_state(inst, definition, cutoff, ["H1"])
plans = []
for q in queries:
    if 'research_' in q['sql']:
        with connection.cursor() as c:
            c.execute("EXPLAIN (ANALYZE,BUFFERS,FORMAT JSON) " + q['sql'])
            plans.append({"sql":q['sql'],"plan":c.fetchone()[0]})
assert len(result[0]['event_state']['events']) == 199
assert result[0]['macro_regime']['by_currency']['USD']['direction'] == 'tightening'
print(json.dumps({"observations":3501,"event_vintages":201,"macro_observations":200,"build_seconds":elapsed,"queries":len(queries),"payload_bytes":len(canonical_json(result[0]).encode()),"manifest_bytes":len(canonical_json(result[2]).encode()),"evidence_bytes":len(canonical_json(result[4]).encode()),"plans":plans},indent=2))
