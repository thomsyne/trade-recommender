import json
import os
import resource
import sys
import time
from datetime import UTC, datetime, timedelta
from decimal import Decimal
from unittest.mock import patch

sys.path.insert(0, os.getcwd())
import django
django.setup()
from django.db import connection
from django.test.utils import CaptureQueriesContext
from market.quality import registered_successor
from market.services import live_candle_completion, store_ingestion
from market.state.canonical import canonical_json
from market.state.compute import build_market_state, compute_market_state, ensure_descriptor_definition
from market.state.manifest import eligible_observations
from market.tests.factories import candle
from market.tests.test_live_observations import make_market

instrument, source = make_market()
definition = ensure_descriptor_definition()
starts = [datetime(2025, 1, 6, 8, tzinfo=UTC)]
for _ in range(3500):
    starts.append(registered_successor(starts[-1], "H1"))
previous = 0
reports = []
for count in (200, 3501):
    values = []
    for i in range(previous, count):
        price = Decimal("1.10") + Decimal(i % 20) / 10000
        changes = {}
        for side, spread in (("bid", Decimal(0)), ("ask", Decimal("0.0002"))):
            for part, change in (("open", 0), ("high", Decimal("0.0012")), ("low", Decimal("-0.0009")), ("close", Decimal("0.0001"))):
                changes[f"{side}_{part}"] = price + spread + change
        values.append(candle(starts[i], **changes))
    cutoff = live_candle_completion(starts[count-1], "H1") + timedelta(seconds=1)
    with patch("market.services.timezone.now", return_value=cutoff):
        store_ingestion(source, instrument, "H1", starts[previous], cutoff, values, {"batch": f"performance-{count}", "requests": []})
    with connection.cursor() as c:
        c.execute("ANALYZE market_candleobservation")
    began = time.perf_counter()
    with CaptureQueriesContext(connection) as queries:
        output = build_market_state(instrument, definition, cutoff, ["H1"])
    elapsed = time.perf_counter() - began
    began = time.perf_counter()
    with CaptureQueriesContext(connection) as write_queries:
        snapshot, _ = compute_market_state(instrument, definition, cutoff, ["H1"])
    write_elapsed = time.perf_counter() - began
    with CaptureQueriesContext(connection) as old_queries:
        old = eligible_observations(instrument, "H1", starts[0] - timedelta(days=1), lookback=2)
    plans = []
    for q in [*queries, *old_queries]:
        if q["sql"].startswith("SELECT"):
            with connection.cursor() as c:
                c.execute("EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) " + q["sql"])
                plans.append({"sql": q["sql"], "plan": c.fetchone()[0]})
    storage = {}
    with connection.cursor() as c:
        for table in ("market_marketstatesnapshot", "market_candleobservation"):
            c.execute("SELECT pg_relation_size(%s), pg_indexes_size(%s), pg_total_relation_size(%s)", [table]*3)
            storage[table] = dict(zip(("heap", "indexes", "total_with_toast"), c.fetchone()))
        c.execute("SELECT pg_column_size(output_payload),pg_column_size(input_manifest),pg_column_size(evidence_manifest) FROM market_marketstatesnapshot WHERE id=%s", [snapshot.pk])
        columns = c.fetchone()
    reports.append({"observations": count, "build_seconds": elapsed, "build_queries": len(queries), "compute_persist_verify_seconds": write_elapsed, "compute_persist_verify_queries": len(write_queries), "manifest_entries": len(output[2]), "manifest_bytes": len(canonical_json(output[2]).encode()), "payload_bytes": len(canonical_json(output[0]).encode()), "evidence_bytes": len(canonical_json(output[4]).encode()), "process_peak_rss_bytes_macos_including_ingestion": resource.getrusage(resource.RUSAGE_SELF).ru_maxrss, "storage_bytes": storage, "snapshot_column_bytes_payload_manifest_evidence": columns, "old_cutoff_rows": len(old), "plans": plans})
    previous = count
print(json.dumps({"scope": "Local synthetic USD_CAD H1; no research fixture rows; not production capacity", "wal": "unmeasured: other disposable databases share this cluster", "reports": reports}, indent=2))
