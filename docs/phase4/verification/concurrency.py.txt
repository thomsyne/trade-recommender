import json
import os
import sys
from concurrent.futures import ThreadPoolExecutor
from datetime import UTC, datetime
from threading import Barrier
from uuid import uuid4

sys.path.insert(0, os.getcwd())
import django
django.setup()
from django.db import connection, connections
from market.models import Instrument, MarketStateDefinition, MarketStateSnapshot
from market.state.compute import compute_market_state, ensure_descriptor_definition

definition_id = ensure_descriptor_definition().pk
instrument_id = Instrument.objects.get(code="USD_CAD").pk
role = "p4_concurrent_" + uuid4().hex
with connection.cursor() as c:
    c.execute(f'CREATE ROLE "{role}" NOSUPERUSER NOLOGIN')
    c.execute(f'GRANT USAGE ON SCHEMA public TO "{role}"')
    c.execute(f'GRANT SELECT, INSERT ON ALL TABLES IN SCHEMA public TO "{role}"')
    c.execute(f'GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO "{role}"')
barrier = Barrier(6)

def worker(_):
    try:
        with connection.cursor() as c:
            c.execute(f'SET ROLE "{role}"')
            c.execute("SELECT rolsuper FROM pg_roles WHERE rolname=current_user")
            assert c.fetchone()[0] is False
        instrument = Instrument.objects.get(pk=instrument_id)
        definition = MarketStateDefinition.objects.get(pk=definition_id)
        barrier.wait(timeout=30)
        snapshot, created = compute_market_state(instrument, definition,
                                                 datetime(2028, 1, 5, tzinfo=UTC), ["H1"])
        return snapshot.pk, snapshot.output_sha256, created
    finally:
        connections.close_all()

try:
    with ThreadPoolExecutor(max_workers=6) as pool:
        results = list(pool.map(worker, range(6)))
    assert len({r[0] for r in results}) == 1
    assert len({r[1] for r in results}) == 1
    assert sum(r[2] for r in results) == 1
    assert MarketStateSnapshot.objects.filter(pk=results[0][0]).count() == 1
    print(json.dumps({"workers": 6, "unique_snapshots": 1, "created": 1,
                      "identical_output": True, "non_superuser": True}))
finally:
    with connection.cursor() as c:
        c.execute(f'DROP OWNED BY "{role}"')
        c.execute(f'DROP ROLE "{role}"')
